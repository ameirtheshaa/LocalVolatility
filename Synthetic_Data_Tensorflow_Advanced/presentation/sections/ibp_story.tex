\begin{frame}{Nicolas's concern (integration by parts)}
    Mean of the NN-implied risk-neutral density:
    \[
        e^{rT}\int_0^\infty K\,\frac{\partial^2 C_{\mathrm{NN}}}{\partial K^2}\,dK
    \]
    Integration by parts (with $C\to 0$ as $K\to\infty$):
    \[
        = e^{rT}\,C_{\mathrm{NN}}(0,T) = \mathbb{E}[S_T]
    \]
    Under no-arbitrage, all three should equal the forward $S_0 e^{rT}$.
    \vspace{0.2cm}
    \begin{block}{Question}
        Where does the visible KDE--NN deviation come from if these means agree?
    \end{block}
\end{frame}

\begin{frame}{Three-way diagnostic test}
    Compute separately at each maturity $T$:
    \begin{center}

            \midrule

            (ii)  & $e^{rT}\,C_{\mathrm{NN}}(0,T)$ \\
            (iii) & $\mathbb{E}[S_T]$ from MC with $\sigma_{\mathrm{NN}}(t,S)$ \\
            ref   & $S_0 e^{rT}$ \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vspace{0.15cm}
    {\small Implemented in \texttt{compute\_mean\_diagnostics}; yellow box on Panel~3. For (i) use \textbf{unclipped} $\partial^2 C/\partial K^2$ (no normalise-to-1).}
\end{frame}

\begin{frame}{Finding 1: IBP holds in our code}
    At every $T$, \textbf{(i) $\equiv$ (ii)} to machine precision ($\sim 6\times 10^{-5}$ relative).
    \vspace{0.3cm}
    \begin{itemize}
        \item Autodiff and numerical integration are consistent
        \item The breakdown is \textbf{not} a wrong $\partial^2 C/\partial K^2$ formula
    \end{itemize}
\end{frame}

\begin{frame}{Solution: $K=0$ boundary penalty + wider strikes}
    \begin{itemize}
        \item Penalty $|C_{\mathrm{NN}}(0,T)-S_0|$ in \texttt{loss\_phi\_cal} ($\lambda_{K0}=10$)
        \item Training strikes $K_{\min}=100$ (was 500)
        \item Retrain $\rightarrow$ \texttt{models/runs/k0\_bc\_retrain}
    \end{itemize}
\end{frame}

\begin{frame}{Summary}
    \begin{itemize}
        \item \textbf{IBP autodiff $+$ integration are exact:} $M_1 \equiv M_2$ to $<\!0.05\%$ at every $T$.
        \item \textbf{NN$_\eta$ MC is well-calibrated:} $M_3$ within $0.7\%$ of $S_0 e^{rT}$ at every $T$.
        \medskip
        \item[\textcolor{ntugold}{\textbullet}] \textbf{Diagnostic fix.} The earlier \texttt{mean\_mc} values \\
              ($1042.19$ at $T=0.5$, etc.) were the in-window MC mean: \\
              \texttt{create\_enhanced\_pdf\_analysis} inadvertently passed the \\
              $[K_{\min},K_{\max}]$-trimmed sample to the IBP $M_3$ computation. \\
              The MC develops a heavy right tail under the local-vol surface, so \\
              asymmetric trimming biased $M_3$ upward by $\sim\!2\%$. Patched at \\
              \texttt{dupire\_pipeline.py:1590} (\texttt{mc\_used}~$\to$~\texttt{mc\_samples}); \\
              the corrected unconditional mean is $1018.25$.
        \medskip
        \item \textbf{Residual KDE-vs-NN deviation:} dominated by NN$_\varphi$ undershoot near $K = 0$ \\
              ($M_1 \approx M_2 \approx 7\%$ below forward).
        \item \textbf{Fix:} K$=$0 boundary penalty in the loss (\texttt{models/runs/k0\_bc\_retrain}) brings all three legs within $\sim\!1\%$ at $T \in \{0.5, 1.0\}$.
    \end{itemize}
\end{frame}
